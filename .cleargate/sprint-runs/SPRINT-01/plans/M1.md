export const ExitCode = { OK: 0, ERROR: 1, USAGE: 2, NEEDS_APPROVAL: 3, DENIED: 4, NEEDS_INPUT: 5 } as const;

export class WikeeperError extends Error {

### C1 — `src/core/output.ts`
```ts
import type { Command } from "commander";
import type { ExitCode } from "./exit-codes.js";
export interface TextSink { write(chunk: string): unknown }
export interface CliIO {
  stdout: TextSink;
  stderr: TextSink;
  stdin: NodeJS.ReadableStream & { isTTY?: boolean };
  cwd: string;                       // commands resolve the repo from ctx.cwd, never process.cwd()
}
export interface OutputContext extends CliIO {
  json: boolean;                     // --json (global, via optsWithGlobals())
  isTTY: boolean;                    // Boolean(stdin.isTTY); "interactive" = ctx.isTTY && !ctx.json
  command: string;                   // "init", "hook guard"; "wikeeper" before dispatch
  setExitCode(code: number): void;   // non-zero exit after a successful emitResult (doctor, hook guard)
}
export interface ResultInput<T = unknown> {
  command: string; data: T; message?: string | readonly string[]; warnings?: readonly string[];
}
export interface SuccessEnvelope<T = unknown> { ok: true; command: string; data: T; warnings: string[] }
export interface ErrorEnvelope { ok: false; command: string; error: { code: string; message: string; details: unknown } }
export function emitResult<T>(ctx: OutputContext, result: ResultInput<T>): void;              // sync
export function emitError(ctx: OutputContext, error: unknown): ExitCode;                       // sync; returns the exit code
export function defaultIO(): CliIO;                                                           // process streams + process.cwd()
export function bindProgramIO(program: Command, io: CliIO): void;                             // used by createProgram only
export function getOutputContext(command: Command): OutputContext;                            // every action handler calls this
```
Behaviour:
- `emitResult`, JSON mode: writes exactly one line, `JSON.stringify({ ok: true, command, data: data ?? null, warnings: warnings ?? [] }) + "\n"`, with keys in that order. Human mode: writes `message` lines (array joined by `\n`, plus a trailing `\n`) to stdout, and each warning as `warning: <w>\n` to stderr. `data` is not printed in human mode.
- `emitError` maps errors in this order:

  | Error | Exit | `error.code` | Human output |
  |---|---|---|---|
  | `CommanderError` with `exitCode === 0` (help, version) | 0 | — | nothing (writes nothing in either mode) |
  | Other `CommanderError` | 2 | `"usage"` | nothing (commander already printed) |
  | `WikeeperError` | its `exitCode` | its `code` | `error: <message>` on stderr |
  | `Error` with a string `code` | 1 | that `code` | `error: <message>` on stderr |
  | Anything else | 1 | `"error"` | `error: <message>` on stderr |

  For `CommanderError`, strip the leading `error: ` from the message. In JSON mode it writes `{ ok: false, command: ctx.command, error: { code, message, details: details ?? null } }`.
- Each action calls `emitResult` once, as its last statement. To fail, an action throws (`WikeeperError`). It never calls `emitError` or `process.exit`.

### C1 — `src/program.ts`
```ts
import { Command } from "commander";
import type { CliIO } from "./core/output.js";
import type { ExitCode } from "./core/exit-codes.js";
export const VERSION: string;                                   // existing, unchanged
export interface CommandModule { register(program: Command): void }
export const defaultCommands: CommandModule[] = [];
export function createProgram(commands: readonly CommandModule[] = defaultCommands, io: CliIO = defaultIO()): Command;
export interface RunCliOptions { commands?: readonly CommandModule[]; io?: Partial<CliIO> }
export function runCli(argv: readonly string[], options?: RunCliOptions): Promise<number>; // argv in user form (no node/script)
```
`createProgram` runs these steps in this order, which matters (see gotchas):
1. `new Command().name("wikeeper")…version(VERSION, "-v, --version")`.
2. `.option("--json", "machine-readable output")`.
3. `.exitOverride()`.
4. `.configureOutput({ writeOut → io.stdout, writeErr → io.stderr, outputError: (s, w) => { if (!jsonMode) w(s) } })`.
5. `bindProgramIO(program, io)`.
6. A root `preAction` hook that records the action command's name for `ctx.command`.
7. The existing root `.action(() => program.outputHelp())`.
8. **Then** `for (const m of commands) m.register(program)`.

`runCli`:
1. Merges `io` over `defaultIO()`.
2. Pre-scans argv for `--json` before any `--`, so JSON mode is known before commander prints a usage error.
3. Calls `parseAsync([...argv], { from: "user" })`.
4. On success, returns the code passed to `setExitCode`, or 0.
5. On a throw, returns `emitError(getOutputContext(program), err)`.

`src/cli.ts` becomes: `process.exitCode = await runCli(process.argv.slice(2));` and never calls `process.exit()`.

### Command-registration pattern (every M2–M4 command story)
```ts
// src/commands/init.ts
import type { Command } from "commander";
import type { CommandModule } from "../program.js";          // type-only: no runtime import cycle
import { emitResult, getOutputContext } from "../core/output.js";
import { ExitCode, WikeeperError } from "../core/exit-codes.js";
export const initCommand: CommandModule = {
  register(program: Command): void {
    program.command("init").description("…").option("--profile <name>", "…", "management")
      .action(async (opts: { profile: string }, cmd: Command) => {
        const ctx = getOutputContext(cmd);                     // json, isTTY, cwd, streams
        if (opts.profile !== "management") throw new WikeeperError(ExitCode.USAGE, "usage", "…");
        emitResult(ctx, { command: "init", data: { created: [], skipped: [] }, message: "…" });
      });
  },
};
```
The insertion each later story makes in `src/program.ts` is one import line plus one array line:
- `import { initCommand } from "./commands/init.js";`, appended to the command-import region.
- `  initCommand,`, appended as the last element of `defaultCommands`.

STORY-002-04 adds one side-effect line instead: `import "./agents/index.js";`.

Rules for every command module:
- Register through `program.command(…)`, never `addCommand(new Command())`. Only `.command()` copies the exitOverride and output configuration.
- Never declare a subcommand-level `--json` option.
- Read JSON mode only from `getOutputContext(cmd).json`.
- Modules are stateless: `register` may be called on many programs (tests).

### C3 — `src/core/templates.ts` (STORY-001-03)
```ts
export const PACKAGE_NAME = "wikeeper";
export interface CopyTemplateResult { written: boolean; sha256: string }  // sha256 of the template bytes (= dest bytes)
export function findPackageRoot(fromUrl: string = import.meta.url): string;      // sync
export function templatesDir(fromUrl?: string): string;                          // sync; additive optional param
export function listTemplateFiles(subdir: string): Promise<string[]>;
export function copyTemplate(rel: string, dest: string): Promise<CopyTemplateResult>;
```
- `findPackageRoot` accepts a `file:` URL or a plain path. It walks `dirname` until it finds a `package.json` that parses with `name === "wikeeper"`. At the filesystem root (`dirname(d) === d`) it throws an `Error` with `code: "E_PACKAGE_ROOT_NOT_FOUND"`.
- `templatesDir(u)` is `path.join(findPackageRoot(u), "templates")`.
- `listTemplateFiles(subdir)` returns regular files only, recursively, as posix paths **relative to `templatesDir()`, including the subdir prefix** (e.g. `["system/SETUP.md", "system/WIKI.md"]`). The output feeds `copyTemplate` directly. Sorting uses code-unit order (`a < b`), not `localeCompare`. It skips `.DS_Store` and `Thumbs.db`. A missing subdir throws `code: "E_TEMPLATE_NOT_FOUND"`.
- `copyTemplate(rel, dest)`:
  - `rel` is a posix path relative to `templatesDir()`; `dest` is an absolute OS path.
  - It creates parent directories and reads the source as a Buffer.
  - If `dest` already holds identical bytes it does not write (`written: false`; mtime is kept), which gives idempotency to 001-04, 001-05 and 002-03.
  - A missing source throws `E_TEMPLATE_NOT_FOUND`.

### C3 — `src/core/manifest.ts` (STORY-001-03, all async)
```ts
export type ManifestKind =
  | "system" | "file" | "dir" | "block" | "skill" | "subagent" | "link" | "gitignore" | "hook" | "rule";
export interface ManifestEntry {
  path: string;                 // posix, relative to repo root, no "./"
  sha256: string | null;        // lowercase hex of the bytes on disk; null when wikeeper does not own the whole content
  kind: ManifestKind;
  agent?: string;               // owning agent: skill | subagent | link | hook | rule
  mode?: "symlink" | "copy";    // skill | link (002-03 R2)
  created?: boolean;            // block | gitignore | hook | rule | dir: wikeeper created the file/dir (remove deletes only if created)
  identity?: string;            // hook | rule: entry identity inside a shared settings file (002-06)
}
export interface Manifest { package_version: string; files: ManifestEntry[] }
export type DriftStatus = "ok" | "modified" | "missing";
export interface ManifestDiffEntry extends ManifestEntry { status: DriftStatus }
export function sha256File(path: string): Promise<string>;                          // streams via node:crypto
export function serializeManifest(manifest: Manifest): string;                     // additive; sync, pure
export function writeManifest(manifest: Manifest, path: string): Promise<boolean>; // true = bytes written
export function readManifest(path: string): Promise<Manifest | null>;              // ENOENT → null
export function diffManifest(manifest: Manifest, root: string): Promise<ManifestDiffEntry[]>;
```
**Kind usage for later stories.** The type permits every kind; each owner story's plan finalises its rows.

| Kind | Story | `sha256` |
|---|---|---|
| `system` | 001-04, 001-05 (`.wiki/system/**`) | required |
| `gitignore` | 001-04, 001-06 (`.gitignore`, with `created`) | null |
| `block` | 002-04 (agent files, with `created`) | null |
| `skill` / `subagent` | 002-03 (file copies) | written bytes |
| `link` | 002-03 (Claude skill symlinks, `mode`) | null |
| `dir` | 002-04 (created dirs) | null |
| `hook` / `rule` | 002-06 (`identity`, `created`) | null |
| `file` | reserved for other whole-file package writes | — |

**Serialisation.**
- `files` is sorted by `path`, then `kind`, then `identity ?? ""` (code-unit order) on a copy; the input is not mutated.
- Each entry is rebuilt with the fixed key order `path, sha256, kind, agent, mode, created, identity`, omitting undefined keys. The top level is `package_version, files`.
- Output is `JSON.stringify(obj, null, 2) + "\n"` with no timestamps.
- `writeManifest` runs `mkdir -p` on the parent and skips the write when the bytes are identical.

**Validation (`readManifest`).** Bad JSON or a bad shape throws `code: "E_MANIFEST_INVALID"`. A bad shape is any of: `package_version` not a string, `files` not an array, an entry `path` not a string, `sha256` neither null nor 64-char hex, or `kind` not in the union. Unknown extra keys are dropped.

**Drift (`diffManifest`).** Entries come back in manifest order. Each path resolves as `path.join(root, ...entry.path.split("/"))`.
- With `sha256`: `missing` on lstat ENOENT; `modified` when the hash differs or the path is a directory; otherwise `ok`.
- With `sha256: null`: `ok` if lstat succeeds, else `missing`. A dangling symlink therefore counts as present; broken-link detection belongs to 002-04's doctor.

### STORY-002-01 — Block injection engine
STORY-002-01 advances the goal by writing the WIKI block with "no byte outside the markers changes" and making `remove` leave no block (Goal check 1, 2, 3).
- **Files (§3.1):** `src/agents/inject.ts`, `templates/agent-block.md` (§6.5 pointer block verbatim as the placeholder body, both markers, LF, trailing newline), `test/agents/inject.test.ts` (all new).
- **Red tests:** `test/agents/inject.red.node.test.ts`. Scenarios:
  1. Plain file → starts with `WIKI_START_V1`, then a blank line, then `# My rules\n`; status `injected`.
  2. Outline, inject→strip byte-identical:
     - LF: block at the top.
     - CRLF: block uses CRLF.
     - BOM + LF: block right after the BOM.
     - Frontmatter: block right after the closing `---`.
     - Empty string: block plus a trailing EOL.
  3. An existing `<!-- WIKI:START v0` block → byte-identical, `skipped`.
  4. Missing file → created with block plus EOL. `stripFromFile` on another missing file → `absent`, and the file is not created.
  5. START without END → byte-identical, `malformed`.
- **Unit tests (≥8):** add a mixed-EOL file (style of the first line break) and frontmatter without a closing `---` (treated as no frontmatter).
- **Algorithm (byte-exact):**
  1. Decode with `new TextDecoder("utf-8", { fatal: true, ignoreBOM: true })`. That keeps U+FEFF as a char. On invalid UTF-8, throw `code: "E_NOT_UTF8"` and leave the file untouched. Never `Buffer.toString` without fatal checking, and never a default `TextDecoder` (it strips the BOM).
  2. Detect the EOL from the first `\n` (preceded by `\r` → CRLF); with no line break, use LF.
  3. Normalise the block with `block.replace(/(\r?\n)+$/, "").split(/\r?\n/).join(eol)`.
  4. Frontmatter applies only when the first line, minus a trailing `\r`, is exactly `---`. The insertion point follows the EOL of the next line that is exactly `---`.
  5. Build `head + block + eol + (rest === "" ? "" : eol + rest)`.
  6. Strip: find the START line with `/^<!-- WIKI:START\b.*-->$/m`. JS `$` with the `m` flag matches before `\r`; cover it with a CRLF test. Then find the first `/^<!-- WIKI:END -->$/m` line after it. Remove START..END plus the END line's own `\r?\n`, then at most one following empty line (`\r?\n`). Only the first block is stripped.
  7. A degenerate frontmatter whose closing `---` has no EOL: inject adds an EOL after `---`, and strip does not remove it. Document this in one test; it is not a Gherkin variant.
- **File helpers:**
  - Write `<real>.wikeeper-tmp`, then `rename`. On failure, unlink the tmp file.
  - `<real>` is `await realpath(path)` when the file exists. Users often symlink `CLAUDE.md → AGENTS.md`, and a rename onto the link would replace it with a regular file.
  - The parent directory is not created.
- **Fixture rule:** build CRLF and BOM fixtures inline as string literals (`"a\r\nb"`, `"﻿…"`). Never commit them as files: `* text=auto eol=lf` (001-01) rewrites them on checkout.
- **Risk:** the block parameter must come from `templates/agent-block.md`. This module never reads templates; 002-04 loads it via `templatesDir()`.
